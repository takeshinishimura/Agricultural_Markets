library(readr)
library(tidyr)
library(dplyr)
library(ggplot2)
library(gghighlight)
library(ggsci)
library(gt)

figdir <- "../fig"
dir.create(figdir, showWarnings = FALSE)

d <- read_csv("../data/卸売市場経由率の推移.csv")

d_long <- d |>
  pivot_longer(
    cols = -年度,
    names_to = "name",
    values_to = "value"
  ) |>
  mutate(name = factor(name, levels = unique(name)))

g <- ggplot(
    data = d_long,
    aes(x = 年度, y = value, colour = name)
  ) +
  geom_line() +
  geom_point() +
  scale_y_continuous(limits = c(0, 100)) +
  scale_color_d3() +
  labs(
    title = "卸売市場経由率の推移",
    x = "年度", 
    y = "卸売市場経由率（%）",
    caption = "出典：農林水産省(2025)「令和6年度 卸売市場データ集」") +
  theme_bw() +
  theme(
    legend.position = "none",
    text = element_text(family = "Hiragino Sans")
  ) +
  gghighlight(
    min(value) > 0,
    line_label_type = "ggrepel_text",
    label_params = list(family = "Hiragino Sans")
  )

ggsave(
  "wholesale_market_trends.pdf",
  plot = g,
  device = cairo_pdf,
  path = figdir,
  width = 10,
  height = 5
)
