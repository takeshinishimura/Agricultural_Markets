1 小売価格データ

ノート総務省統計局　小売物価統計調査　小売物価統計調査（動向編）

https://www.e-stat.go.jp/stat-search/files?page=1&layout=datalist&toukei=00200571&tstat=000000680001&cycle=1&tclass1val=0

小売物価統計調査 小売物価統計調査（動向編）主要品目の都市別小売価格－都道府県庁所在市及び人口15万以上の市(2000年1月～) | 統計表・グラフ表示 | 政府統計の総合窓口 からデータを入手する。

ただし，e-statのAPIは，一度のリクエストで取得できるデータの行数に制限がある場合がある。 この制限を超えるデータを取得するために，ページネーションを利用して複数回のリクエストを行い，それぞれの結果を結合する。

for (item in items) {
cdCat02 = meta_info$cat02$`@code`[grep(item, meta_info$cat02$`@name`)]

item <- gsub("\\\\s|\\\\|\\$", "", item)
message(item)

# 統計データ取得
df <- estat_getStatsData(
  appId = appId,
  statsDataId = statsDataId,
  cdCat02 = cdCat02
)

df$time <- as.Date(
  lubridate::ymd(paste0(gsub("年", "-", df$`時間軸（月）`), "-01"))
)

# 不要なデータを削除
df <- df |>
  filter(!grepl("1901|1912|1952|2025|2027|2171", 銘柄))
## 1901 緑茶(番茶，上)【2006年12月調査終了】
## 1912 紅茶(輸入品)【2004年12月調査終了】
## 1952 ノンアルコールビール
## 2025 ビール(輸入品)【2006年12月調査終了】
## 2027 ビール風アルコール飲料
## 2171 ビール(外食)

# 全国平均価格を計算
avg_values <- df |>
  group_by(time) |>
  summarise(avg_value = mean(value, na.rm = TRUE))

# df <- left_join(df, avg_values, by = "time")
new_row <- setNames(as.list(rep(NA, length(names(df)))), names(df))
new_row$地域 <- "全国平均"
new_row$value <- avg_values$avg_value
new_row$time <- avg_values$time
df <- rbind(df, data.frame(new_row, check.names = FALSE))

# データ保存
write.table(df, file = file.path(outdir, paste0("retail_price_", item, "_all.csv")), sep = ",", row.names = FALSE)

# 価格推移の折れ線グラフの作図
g <- ggplot(
    data = df,
    aes(x = time, y = value, color = 地域)
  ) +
  geom_line() +
  scale_y_continuous(labels = comma) +
  scale_color_d3() +
  labs(
    title = paste0(item, "の価格推移"),
    y = "価格（円）",
    x = "年",
    color = "",
    caption = "出典：総務省統計局「小売物価統計調査（動向編）」"
  ) +
  theme_bw() +
  theme(
    text = element_text(family = "Hiragino Sans")
  ) +
  gghighlight(
    grepl("松山市|全国平均", 地域),
#   line_label_type = "ggrepel_text",
    label_params = list(family = "Hiragino Sans")
  )

print(unique(df$annotation))
print(g)

ggsave(
  paste0("retail_price_", item, "_all", ".pdf"),
  plot = g,
  device = cairo_pdf,
  path = figdir,
  width = 8,
  height = 4
)


df2 <- df |>
  filter(地域 == "松山市") |>
  mutate(
    year = lubridate::year(time),
#   month = lubridate::month(time, label = TRUE, abbr = TRUE) # 日本語環境だと期待通りにならない
    month = factor(month.abb[lubridate::month(time)], levels = month.abb, ordered = TRUE)
  )

g2 <- ggplot(
    data = df2,
    aes(x = month, y = value, group = year, color = as.factor(year))
  ) +
  geom_line() +
  scale_y_continuous(labels = comma) +
  scale_color_viridis(discrete = TRUE, option = "G") +
  labs(
    title = paste0(item, "の価格推移（松山市）"),
    y = "価格（円）",
    x = "月",
    color = "年",
    caption = "出典：総務省統計局「小売物価統計調査（動向編）」"
  ) +
  theme(
    text = element_text(family = "Hiragino Sans")
  )

print(g2)

ggsave(
  paste0("retail_price_", item, "_yearly.pdf"),

  device = cairo_pdf,
  path = figdir,
  width = 8,
  height = 4
)
